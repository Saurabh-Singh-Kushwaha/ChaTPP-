rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function participant(data) {
      return signedIn() && (
        data.fromUid == request.auth.uid ||
        data.toUid == request.auth.uid
      );
    }

    match /users/{userId} {
      allow read: if signedIn();
      allow create: if isOwner(userId)
        && request.resource.data.uid == userId
        && request.resource.data.keys().hasOnly([
          'uid', 'displayName', 'handle', 'bio', 'photoURL',
          'createdAt', 'updatedAt'
        ]);
      allow update: if isOwner(userId)
        && request.resource.data.uid == resource.data.uid;
      allow delete: if isOwner(userId);
    }

    match /privateUsers/{userId} {
      allow read, create, update, delete: if isOwner(userId);
    }

    match /handles/{handle} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.uid == request.auth.uid;
      allow update: if signedIn()
        && resource.data.uid == request.auth.uid
        && request.resource.data.uid == request.auth.uid;
      allow delete: if signedIn()
        && resource.data.uid == request.auth.uid;
    }

    match /privacyLocks/{userId} {
      allow read, create, update, delete: if isOwner(userId);
    }

    match /friendRequests/{requestId} {
      allow read: if signedIn() && participant(resource.data);

      allow create: if signedIn()
        && requestId == request.resource.data.fromUid + '_' + request.resource.data.toUid
        && request.resource.data.fromUid == request.auth.uid
        && request.resource.data.toUid != request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.keys().hasOnly([
          'fromUid', 'toUid', 'status', 'createdAt'
        ]);

      allow update: if signedIn()
        && resource.data.fromUid == request.resource.data.fromUid
        && resource.data.toUid == request.resource.data.toUid
        && resource.data.status == 'pending'
        && (
          (request.auth.uid == resource.data.toUid
            && request.resource.data.status == 'accepted') ||
          (request.auth.uid == resource.data.fromUid
            && request.resource.data.status == 'cancelled')
        );

      allow delete: if signedIn() && participant(resource.data);
    }

    match /friendships/{friendshipId} {
      allow read: if signedIn() && request.auth.uid in resource.data.users;

      allow create: if signedIn()
        && friendshipId == request.resource.data.users[0] + '_' + request.resource.data.users[1]
        && request.resource.data.users.size() == 2
        && request.auth.uid in request.resource.data.users
        && request.resource.data.users[0] != request.resource.data.users[1]
        && (
          (exists(/databases/$(database)/documents/friendRequests/$(request.resource.data.users[0] + '_' + request.resource.data.users[1]))
            && getAfter(/databases/$(database)/documents/friendRequests/$(request.resource.data.users[0] + '_' + request.resource.data.users[1])).data.status == 'accepted')
          ||
          (exists(/databases/$(database)/documents/friendRequests/$(request.resource.data.users[1] + '_' + request.resource.data.users[0]))
            && getAfter(/databases/$(database)/documents/friendRequests/$(request.resource.data.users[1] + '_' + request.resource.data.users[0])).data.status == 'accepted')
        );

      allow delete: if signedIn() && request.auth.uid in resource.data.users;
      allow update: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
